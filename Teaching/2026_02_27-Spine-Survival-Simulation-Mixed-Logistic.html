set.seed(1)

# --- Define structure ---
n_mice <- 6
n_dendrites_per_mouse <- 4
n_spines_per_dendrite <- 10
days <- 1:8

# --- Mouse-level info ---
mouse_ids <- paste0("mouse_", sample(1000:9999, n_mice))
genotypes <- rep_len(c("wt", "exp"), n_mice)  # alternating, or randomize
mouse_df <- tibble(
  mouse_id = mouse_ids,
  genotype = genotypes
)

# --- Dendrite-level info ---
dendrite_df <- tibble(
  mouse_id = rep(mouse_ids, each = n_dendrites_per_mouse),
  dendrite_id = paste0("dend_", sample(10000:99999, n_mice * n_dendrites_per_mouse))
)

# --- Spine-level skeleton ---
spine_df <- tibble(
  dendrite_id = rep(dendrite_df$dendrite_id, each = n_spines_per_dendrite),
  spine_id = paste0("spine_", sample(100000:999999, n_mice * n_dendrites_per_mouse * n_spines_per_dendrite))
)

# --- Expand to daily observations ---
df <- spine_df %>%
  tidyr::crossing(day = days) %>%
  left_join(dendrite_df, by = "dendrite_id") %>%
  left_join(mouse_df, by = "mouse_id")

# --- Add cut ---
df <- df %>%
  mutate(cut = day >= 5)

# --- Simulate spine presence (survival-style, but with random start) ---
# For each spine: sample a birth day (or never born = Inf), and a death day
# Spine is present (1) between birth and death day, inclusive

spine_survival <- spine_df %>%
  distinct(spine_id) %>%
  mutate(
    birth_day = sample(1:8, n(), replace = TRUE),   # day spine first appears
    # death day is >= birth day; some spines survive the whole window
    lifespan  = sample(1:8, n(), replace = TRUE),   # how many days it lives
    death_day = pmin(birth_day + lifespan - 1, 8)
  )

df <- df %>%
  left_join(spine_survival, by = "spine_id") %>%
  mutate(
    spine = as.integer(day >= birth_day & day <= death_day)
  ) %>%
  select(-birth_day, -lifespan, -death_day)

# --- Add area (only meaningful when spine == 1, but simulate for all) ---
df <- df %>%
  mutate(
    area = rnorm(n(), mean = 1.5, sd = 0.4),
    day  = factor(day)
  )

# --- Final column order ---
df <- df %>%
  select(mouse_id, genotype, dendrite_id, spine_id, day, cut, spine, area)

days <- 1:8
cut_day <- 5

# initialize each spine as alive on day 1 with some probability
spine_states <- spine_df %>%
  mutate(alive = rbinom(n(), 1, 0.5))  # baseline prevalence

all_days <- list()

for (d in days) {

  # define rates
  if (d < cut_day) {
    p_birth <- 0.05
    p_death <- 0.05
  } else {
    p_birth <- 0.05
    p_death <- 0.25   # increased death after cut
  }

  # deaths
  died <- rbinom(nrow(spine_states), 1, p_death)
  spine_states$alive <- ifelse(spine_states$alive == 1 & died == 1, 0, spine_states$alive)

  # births (only for currently dead spines)
  born <- rbinom(nrow(spine_states), 1, p_birth)
  spine_states$alive <- ifelse(spine_states$alive == 0 & born == 1, 1, spine_states$alive)

  all_days[[d]] <- spine_states %>%
    mutate(day = d, spine = alive)
}

df2 <- bind_rows(all_days) %>%
  left_join(dendrite_df, by = "dendrite_id") %>%
  left_join(mouse_df, by = "mouse_id") %>%
  mutate(day = factor(day),
         cut = day %in% 5:8)
df2$area =df$area

Logistic Regression

Here, we specify a logistic regression using glmer with the parameters we discussed

I want to point out something I realized after our discussion, if we consider day as a categorical variable rather than continuous, then day as a factor perfectly covaries with cut, so cut must be removed from the model, two variables cannot perfectly covary. We can still analyze the difference before vs after the cut by just looking at the probability of survival before and after, and we could probably recover the p-value of the cut in some way, though I’d have to think more about it.

Because the spine is the unit of repeated measurement, we should actually include a random effect of spine nested within dendrite (1|dendrite_id/spine_id). If we wanted to be super precise we would do (1|mouse/dendrite/spine), but that may be too conservative statistically.

model = glmer(spine ~genotype*day +(1|dendrite_id/spine_id), data = df2,
              family = 'binomial', 
             control = glmerControl(optimizer = "bobyqa") #I include this optimizer because the model was not able to converge with my fake data, hopefully it does with the real data
